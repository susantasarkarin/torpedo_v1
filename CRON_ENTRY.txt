# 1. Auto-deploy from GitHub every 10 minutes
*/10 * * * * cd /var/www/campaign_platform && ./deploy.sh --no-confirm >> /var/www/campaign_platform/cron-deploy.log 2>&1

# 2. Watchdog: Gmail sync, CPX/Cint health check every 5 minutes (uses venv)
*/5 * * * * cd /var/www/campaign_platform/backend && ./venv/bin/python scripts/watchdog.py >> /var/log/campaign_watchdog.log 2>&1

# 3. Full Gmail sync (all emails) every hour - for catching up (uses venv)
0 * * * * cd /var/www/campaign_platform/backend && ./venv/bin/python scripts/sync_gmail.py >> /var/log/gmail_full_sync.log 2>&1

# ============================================
# NOTES:
# - The watchdog script checks Gmail, CPX, and Cint every 5 minutes
# - If the backend APScheduler jobs are running, these cron jobs provide redundancy
# - The deploy job syncs code from GitHub to keep VM/GitHub aligned
# - All Python scripts use ./venv/bin/python to access installed packages
